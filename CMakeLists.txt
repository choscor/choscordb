cmake_minimum_required(VERSION 3.28)
include(cmake/Version.cmake)
project(ChoscorDB VERSION ${CHOSCORDB_VERSION} LANGUAGES CXX)
if(NOT DEFINED CMAKE_CXX_COMPILER_LAUNCHER)
  set(CMAKE_CXX_COMPILER_LAUNCHER
    "${PROJECT_SOURCE_DIR}/scripts/dev/compiler-cache" CACHE STRING
    "Optional shared compiler cache for C++ builds")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(CTest)
include(cmake/MacOSUpdater.cmake)
include(cmake/WindowsLinuxUpdater.cmake)
include(cmake/BuildPrivacy.cmake)
include(cmake/Rust.cmake)
include(cmake/QualityGates.cmake)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Test Concurrent Svg)
include(cmake/MacOSQtCompatibility.cmake)
choscordb_remove_unavailable_agl()
find_path(QSCINTILLA_INCLUDE_DIR Qsci/qsciscintilla.h REQUIRED)
find_library(QSCINTILLA_LIBRARY NAMES qscintilla2_qt6 qscintilla2-qt6 REQUIRED)
include(cmake/DesktopComponents.cmake)
choscordb_add_library(choscordb-update-metadata desktop/app/update_readiness.cpp)
target_include_directories(choscordb-update-metadata PUBLIC desktop)
target_link_libraries(choscordb-update-metadata PUBLIC choscordb-rust Qt6::Core)
choscordb_add_executable(choscordb WIN32 MACOSX_BUNDLE desktop/app/main.cpp)
target_link_libraries(choscordb PRIVATE choscordb-desktop)
target_compile_definitions(choscordb PRIVATE CHOSCORDB_VERSION="${PROJECT_VERSION}")
set_target_properties(choscordb PROPERTIES
  MACOSX_BUNDLE_BUNDLE_NAME ChoscorDB
  MACOSX_BUNDLE_GUI_IDENTIFIER com.choscor.ChoscorDB
  MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
  MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}")
choscordb_configure_updater(choscordb)
choscordb_configure_windows_linux_updater(choscordb)
include(cmake/MacOSIcon.cmake)
if(BUILD_TESTING)
  install(CODE "message(FATAL_ERROR \"Installation requires BUILD_TESTING=OFF to exclude the developer preview. Use the release preset.\")")
endif()
install(TARGETS choscordb BUNDLE DESTINATION . RUNTIME DESTINATION bin)
if(APPLE)
  install(FILES docs/licenses/shadcn/LICENSE.md DESTINATION "$<TARGET_BUNDLE_DIR_NAME:choscordb>/Contents/Resources/licenses" RENAME LICENSE-SHADCN)
  install(FILES docs/licenses/shadcn/manifest.json DESTINATION "$<TARGET_BUNDLE_DIR_NAME:choscordb>/Contents/Resources/licenses" RENAME SOURCE-SHADCN.json)
  install(FILES desktop/resources/fonts/OFL.txt desktop/resources/fonts/SOURCE-GEIST.json LICENSE desktop/resources/icons/LICENSE-LUCIDE desktop/resources/icons/SOURCE-LUCIDE.json desktop/resources/icons/SOURCE-DATABASE-LOGOS.json DESTINATION "$<TARGET_BUNDLE_DIR_NAME:choscordb>/Contents/Resources/licenses")
  install(DIRECTORY desktop/resources/icons/ DESTINATION "$<TARGET_BUNDLE_DIR_NAME:choscordb>/Contents/Resources/icons" FILES_MATCHING PATTERN "*.svg" PATTERN "*.png")
else()
  install(FILES docs/licenses/shadcn/LICENSE.md DESTINATION share/licenses/choscordb RENAME LICENSE-SHADCN)
  install(FILES docs/licenses/shadcn/manifest.json DESTINATION share/licenses/choscordb RENAME SOURCE-SHADCN.json)
  install(FILES desktop/resources/fonts/OFL.txt desktop/resources/fonts/SOURCE-GEIST.json LICENSE desktop/resources/icons/LICENSE-LUCIDE desktop/resources/icons/SOURCE-LUCIDE.json desktop/resources/icons/SOURCE-DATABASE-LOGOS.json DESTINATION share/licenses/choscordb)
  install(DIRECTORY desktop/resources/icons/ DESTINATION share/choscordb/icons FILES_MATCHING PATTERN "*.svg" PATTERN "*.png")
endif()
option(CHOSCORDB_BUILD_PERFORMANCE_PROBE "Build the standalone desktop performance probe" OFF)
if(CHOSCORDB_BUILD_PERFORMANCE_PROBE)
  choscordb_add_executable(choscordb-desktop-probe tests/performance/desktop_probe.cpp tests/performance/process_memory.cpp)
  target_link_libraries(choscordb-desktop-probe PRIVATE choscordb-desktop)
  if(WIN32)
    target_link_libraries(choscordb-desktop-probe PRIVATE psapi)
  endif()
endif()

option(CHOSCORDB_ENABLE_STAGING "Add the explicit unsigned application staging target" OFF)
if(CHOSCORDB_ENABLE_STAGING)
  include(cmake/Deploy.cmake)
endif()

if(BUILD_TESTING)
  include(tests/Tests.cmake)
endif()

choscordb_add_quality_targets(choscordb-desktop)
