cmake_minimum_required(VERSION 3.28)
project(ChoscorDB VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)

find_library(QSCINTILLA_LIBRARY NAMES qscintilla2_qt6 qscintilla2-qt6 REQUIRED)
add_library(choscordb-widgets desktop/app/query_settings.cpp desktop/widgets/query_settings_dialog.cpp desktop/bridge/template_service.cpp desktop/widgets/editor_completion.cpp desktop/bridge/completion_service.cpp desktop/bridge/result_column_adapter.cpp desktop/app/editor_preferences.cpp desktop/widgets/preferences_dialog.cpp desktop/models/shortcut_catalog.cpp desktop/widgets/search_panel.cpp desktop/widgets/history_dock.cpp desktop/models/history_model.cpp desktop/widgets/sql_editor.cpp desktop/widgets/document_io.cpp desktop/app/main_window.cpp desktop/app/workspace_recovery.cpp desktop/models/result_table_model.cpp desktop/bridge/engine_adapter.cpp desktop/app/query_workspace.cpp desktop/models/navigator_model.cpp desktop/app/navigator_controller.cpp desktop/models/value_preview_model.cpp desktop/widgets/value_detail_dialog.cpp desktop/widgets/export_dialog.cpp desktop/widgets/profile_dialog.cpp)
target_include_directories(choscordb-widgets PUBLIC desktop ${QSCINTILLA_INCLUDE_DIR})
target_link_libraries(choscordb-widgets PUBLIC Qt6::Widgets Qt6::Concurrent ${QSCINTILLA_LIBRARY} choscordb-rust)
if(WIN32)
  target_compile_definitions(choscordb-widgets PUBLIC QSCINTILLA_DLL)
endif()
add_dependencies(choscordb-widgets cargo-build_choscordb_bridge)
add_executable(choscordb WIN32 MACOSX_BUNDLE desktop/app/main.cpp)
target_link_libraries(choscordb PRIVATE choscordb-widgets)
set_target_properties(choscordb PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME ChoscorDB MACOSX_BUNDLE_GUI_IDENTIFIER org.choscordb.desktop)
install(TARGETS choscordb BUNDLE DESTINATION . RUNTIME DESTINATION bin)
if(APPLE)
  install(FILES LICENSE DESTINATION "$<TARGET_BUNDLE_DIR_NAME:choscordb>/Contents/Resources/licenses")
else()
  install(FILES LICENSE DESTINATION share/licenses/choscordb)
endif()
